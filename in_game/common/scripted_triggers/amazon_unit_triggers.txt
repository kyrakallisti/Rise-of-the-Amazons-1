#Unit Triggers
##Hellenic
    unit_heraean_kataphractoi_location_trigger = {
        dominant_culture = { has_culture_group = culture_group:hellenic_amazon_group }
        owner ?= {
            has_primary_or_accepted_culture = culture:heraean_amazon
        }
    }

    unit_norsellenic_aspidoklastai_unit_location_trigger = {
        dominant_culture = {
            has_culture_group = culture_group:hellenic_amazon_group
            has_culture_group = culture_group:norse_amazon_group
        }
        owner ?= {
            has_primary_or_accepted_culture = culture:norsellenic_amazon
        }
    }

    unit_hoplitai_pop_trigger = {
        OR = {
            culture = culture:athenaean_amazon
            AND = {
                location = { 
                    owner ?= { 
                        has_estate_privilege = estate_privilege:hoplite_politeia
                        has_primary_or_accepted_culture = root.dominant_culture
                    }
                    OR = { 
                        integration_level = integrated
                        integration_level = core
                    }
                }
            }
        }
    }
