			if = {
				limit = {
					exists = root

					desc = "HEIR_SELECTION_FEMALE_DESC"

	matriarchal_elective_succession = {
		use_election = yes
		include_ruler_siblings = yes
		all_in_country = yes
		all_in_dynasty = yes
		through_female = yes
		ignore_ruler = yes
		allow_female = yes
		allow_male = no
		term_duration = 0	#0 = until ruler death
		allowed_estates = { nobles_estate }
		heir_is_allowed = {
		}
		potential = {
			culture = { has_culture_group = culture_group:amazon_group }
			has_unlocked_heir_selection_trigger = { type = elective_succession }
		}
		locked = {
			trigger_if = {
				limit = { has_variable = rise_of_szlachta_szlachta_outcome_variable }
				OR = {
					custom_tooltip = {
						text = has_had_struggle_for_royal_power_disaster_tt
						has_variable = rise_of_szlachta_szlachta_outcome_variable
					}
					"estate_power(estate_type:crown_estate)" < 0.5
				}
			}
			trigger_else = { always = no }
		}
		max_possible_candidates = 3
		succession_effect = {
			add_legitimacy = legitimacy_mild_penalty
		}

		calc = {
			if = {
				limit = {
					exists = root
				}
				add = {
					desc = "HEIR_SELECTION_ADM_DESC"
					value = root.adm
				}
				add = {
					desc = "HEIR_SELECTION_DIP_DESC"
					value = root.dip
				}
				add = {
					desc = "HEIR_SELECTION_MIL_DESC"
					value = root.mil
				}
				
				if = {
					limit = { is_female = no }
					add = {
						desc = "HEIR_SELECTION_MALE_DESC"
						value = -100
					}
				}

				if = {
					limit = { exists = root.mother }
					
					if = {
						limit = {
							any_parent = {
								this = scope:target.ruler
							}
						}
						add = {
							desc = "HEIR_SELECTION_PARENT_IS_CURRENT_RULER_DESC"
							value = 100
						}
					}				
					else = {
						if = {
							limit = { 
								scope:target = { 
									any_ruler = { 
										is_alive = no
										root.mother = this 
									} 
								}
							}
							add = {
								desc = "HEIR_SELECTION_MOTHER_WAS_HISTORICAL_RULER_DESC"
								value = 100
							}
						}
					}
				}
				if = { 
					limit = { modifier:blocked_from_being_ruler = yes }
					add = {
						desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
						value = -1000 
					}
				}			
				if = {
					limit = { 
						NOT = {
							scope:target ?= {
								any_primary_or_accepted_culture = { root.culture = this }
							}
						} 
					}
					add = {
						desc = "HEIR_SELECTION_NON_ACCEPTED_CULTURE_DESC"
						value = -100
					}
				}
			}
			if = {
				limit = {
					NOT = { exists = root }
				}
				add = {
					desc = "HEIR_SELECTION_MOTHER_WAS_HISTORICAL_RULER_DESC"
					value = 100
				}
				add = {
					desc = "HEIR_SELECTION_PARENT_IS_CURRENT_RULER_DESC"
					value = 100
				}
				add = {
					desc = "HEIR_SELECTION_ADM_DESC"
					value = 1
				}
				add = {
					desc = "HEIR_SELECTION_DIP_DESC"
					value = 1
				}
				add = {
					desc = "HEIR_SELECTION_MIL_DESC"
					value = 1
				}
				add = { 
					desc = "HEIR_SELECTION_MALE_DESC"
					value = -100
				}
				add = {
					desc = "HEIR_SELECTION_NON_ACCEPTED_CULTURE_DESC"
					value = -100
				}
			}
		}
	}
