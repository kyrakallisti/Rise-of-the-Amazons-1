# A Blessing from the All-Mother
flavor_amz.1 = {
	type = country_event
	title = flavor_amz.1.title
	desc = flavor_amz.1.desc
	historical_info = flavor_amz.1.historical_info

	fire_only_once = yes

	dynamic_historical_event = {
		tag = ABAS
		from = 1337.1.1
		to = 1348.1.1
		monthly_chance = 100
	}

	trigger = {
		character:abas_althaia_hippolytide_5 = {
			is_alive = yes
			owner ?= ROOT
			is_spouse_of = character:anea_sven_svanardottir_1 
		}
		character:abas_agathe_the_bastard = {
			is_alive = yes
		}
	}

	immediate = {
		character:abas_althaia_hippolytide_5 = { save_scope_as = mother_owl }
		character:abas_agathe_the_bastard = { save_scope_as = bastard_owl }
	}

	option = {
		name = flavor_amz.1.a
		custom_tooltip = flavor_amz.1.a.tt
		historical_option = yes
		ai_will_select  = { value = 95 }

		add_stability = stability_weak_penalty
		add_legitimacy = legitimacy_mild_bonus
	}

	option = {
		name = flavor_amz.1.b
		custom_tooltip = flavor_amz.1.b.tt
		ai_will_select  = { value = 5 }

		scope:bastard_owl = {
			if = {
					limit = { has_character_modifier = bastard }
					remove_character_modifier = bastard
				}
			if = {
					limit = { has_dynasty = no }
					change_dynasty = scope:mother_owl.dynasty
				}
			set_nickname = nick_the_ascended
		}

		c:ANEA = {
			add_opinion = {
				target = root
				modifier = opinion_insult
			}
		}

		add_stability = stability_mild_bonus
		add_legitimacy = legitimacy_extreme_penalty
	} 

	after = {
		hidden_effect = {
			character:anea_sven_svanardottir_1 = {
				impregnate = scope:mother_owl
			}
			trigger_event_non_silently = {
				id = flavor_amz.2
				months = 6
			}			
		}
	}
}

# The Return of the Queen
flavor_amz.4 = {
	type = country_event
	title = flavor_amz.4.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					c:ANEA = { has_country_modifier = defenders_of_the_child }
				}
				desc = flavor_amz.4.land.desc
			}
			triggered_desc = {
				trigger = {
					NOT = { c:ANEA = { has_country_modifier = defenders_of_the_child } }
				}
				desc = flavor_amz.4.sea.desc
			}
		}
	}
	historical_info = {
		first_valid = {
			triggered_desc = {
				trigger = {
					c:ANEA = { has_country_modifier = defenders_of_the_child }
				}
				desc = flavor_amz.4.land.historical_info
			}
			triggered_desc = {
				trigger = {
					NOT = { c:ANEA = { has_country_modifier = defenders_of_the_child } }
				}
				desc = flavor_amz.4.sea.historical_info
			}
		}
	}

	fire_only_once = yes

	dynamic_historical_event = {
		tag = ANEA
		from = 1350.1.1
		to = 1400.1.1
		monthly_chance = 100
	}

	trigger = {
		character:abas_elpis_hippolytide_child = { 
			is_alive = yes 
			is_adult = yes
		}
	}

	immediate = {	
		character:abas_elpis_hippolytide_child = { save_scope_as = baby_owl }
		if = {
			limit = {
				c:ANEA = { has_country_modifier = defenders_of_the_child }
			}
			scope:baby_owl = {
				add_trait = trait:defensive_planner
				add_trait = trait:righteous
				add_adm = {
					value = 50
					max = 88
				}
				add_dip = {
					value = 50
						max = 64
					}
				add_mil = {
					value = 50
					max = 80
				}
			}
		}
		else = {
			scope:baby_owl = {
				add_trait = trait:naval_showman
				add_trait = trait:free_thinker
				add_adm = {
					value = 50
					max = 60
				}
				add_dip = {
					value = 80
					max = 92
				}
				add_mil = {
					value = 50
					max = 80
				}
			}
		}
	}

	option = {
		name = flavor_amz.4.a
		custom_tooltip = flavor_amz.4.a.tt
		###ai_will_select = { value = 100 }
		###historical_option = yes

		set_new_ruler = scope:baby_owl
		set_country_rank = country_rank:rank_kingdom

		add_legitimacy = legitimacy_mild_bonus
		add_prestige = prestige_severe_bonus

		hidden_effect = {
			change_country_flag = ANEA_Basilissa	

			c:ABAS = {
				every_core_location = {
					add_core = c:ANEA
				}
			}
			c:ANEA = {
				every_core_location = {
					add_core = c:ABAS
				}
			}

			every_country = {
				limit = {
					capital ?= { region = region:motherland_region }
					culture = { has_culture_group = culture_group:norse_amazon_group }
					NOR = { 
						this = c:ANEA 
						this = c:ABAS 
					}
				}
				add_opinion = { target = c:ANEA modifier = opinion_the_return_of_the_queen }
				add_opinion = { target = c:ABAS modifier = opinion_pretender }
			}
			every_country = {
				limit = {
					capital ?= { region = region:motherland_region }
					culture = culture:heraean_amazon
					NOR = { 
						this = c:ANEA 
						this = c:ABAS 
					}
				}
				add_opinion = { target = c:ANEA modifier = opinion_pretender }
				add_opinion = { target = c:ABAS modifier = opinion_defender_of_hellenism }
			}
		}

		c:ABAS = {
			every_disloyal_subject = {
				limit = {
					capital ?= { region = region:motherland_region }
					culture = { has_culture_group = culture_group:norse_amazon_group }
					NOT = { this = c:ANEA }
				}
				make_subject_of = {
					target = root
					type = subject_type:vassal
				}
			}
		}

		add_casus_belli = {
			target = c:ABAS
			type = casus_belli:cb_claim_throne
			years = 10
		}	
		c:ABAS = {
			add_casus_belli = {
				target = c:ANEA
				type = casus_belli:cb_subjugation
				years = 10
			}
		}
	}
}
